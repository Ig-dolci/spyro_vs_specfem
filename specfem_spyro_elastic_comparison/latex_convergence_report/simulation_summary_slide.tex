        \includegraphics[width=\linewidth]{../results/spyro_3d_receiver_convergence_top_free_dirichlet/comparison/spatial_convergence_no_timing.png}
    \end{column}
    \begin{column}{0.40\textwidth}
        \scriptsize
        \textbf{Setup:}\\
        Topo livre + Dirichlet, $h_{\text{ref}}=0.07$\\
        41 receptores, $t_f=0.4$\,s, $\Delta t=6\times10^{-4}$\,s\\[0.5em]
        \textbf{Resultado:}\\
        $p_{ux}\approx1.72$, $p_{uy}\approx1.85$, $p_{uz}\approx2.07$\\[0.5em]
        Curvas abaixo de $O(h^5)$ --- dominadas por proje\c{c}\~ao/interpola\c{c}\~ao e desalinhamento receptores--n\'os.
    \end{column}
\end{columns}
